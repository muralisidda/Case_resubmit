DECLARE @CaseNum VARCHAR(50) = '62738806';
DECLARE @InvestorId VARCHAR(50);
DECLARE @CaseStarted DATETIME;
DECLARE @ProcId INT;
DECLARE @ProcName VARCHAR(255);

-- Get case details
SELECT
    @InvestorId = cda.field_value,
    @CaseStarted = cia.started,
    @ProcId = cia.proc_id,
    @ProcName = pi.proc_name -- replace if different
FROM tibcodomain_rep.swpro.case_data cda WITH (NOLOCK)
INNER JOIN tibcodomain_rep.swpro.case_information cia WITH (NOLOCK)
    ON cia.casenum = cda.casenum
INNER JOIN tibcodomain_rep.swpro.proc_index pi WITH (NOLOCK)
    ON pi.proc_id = cia.proc_id
WHERE cda.casenum = @CaseNum
  AND cda.field_name = 'IVINVESTORID';

DECLARE @SQL NVARCHAR(MAX);

SET @SQL = '
SELECT TOP 1
    ''' + @CaseNum + ''' AS CaseNum,
    ''' + @InvestorId + ''' AS InvestorId,
    ' + CAST(@ProcId AS VARCHAR(20)) + ' AS ProcId,
    ''' + @ProcName + ''' AS ProcName,
    ''' + CONVERT(VARCHAR(23), @CaseStarted, 121) + ''' AS CaseStarted,
    MessageIdentifier,
    CreateDateTime,
    ABS(DATEDIFF(SECOND, CreateDateTime,
                 ''' + CONVERT(VARCHAR(23), @CaseStarted, 121) + ''')) AS DiffSeconds
FROM OPENQUERY([AJB10VSS01\AJB10VSS01], ''
    SELECT
        ClientIdentifier,
        MessageIdentifier,
        CreateDateTime
    FROM websupportdatabase.dbo.wsd_messages WITH (NOLOCK)
    WHERE ClientIdentifier = ''''' + @InvestorId + '''''
'')
ORDER BY DiffSeconds;';

EXEC(@SQL);
